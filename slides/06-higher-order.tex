\begin{frame}[fragile]
  \frametitle{Higher-order functions}
  \begin{itemize}
  \item Functions are first-class citizens in Lean
  \item A function can be
    \begin{itemize}
    \item stored in data
    \item argument of a (\textbf{higher-order}) functions
    \item returned from a function
    \end{itemize}
  \end{itemize}
\end{frame}
%----------------------------------------------------------------------
\begin{frame}[fragile]
  \frametitle{Examples of higher-order functions}

  \begin{block}<+->{Most higher-order functions are polymorphic}
\begin{lstlisting}
List.map      : (α → β) → List α → List β 
List.filter  : (α → Bool) → List α → List α
\end{lstlisting}
  \end{block}
  \begin{block}<+->{Example uses}
\begin{lstlisting}
#eval List.map even [1, 2, 3, 4]
#eval List.filter even [1, 2, 3, 4]
\end{lstlisting}
  \end{block}
  \begin{alertblock}<+->{Lean: type quantifiers are implicit}
\begin{lstlisting}
List.map   {α β}  : (α → β) → List α → List β 
List.filter {α}  : (α → Bool) → List α → List α
\end{lstlisting}
  \end{alertblock}
\end{frame}
\begin{frame}[fragile]
  \frametitle{Instantiation}
  \begin{block}<+->{Instances of a polymorphic type}
    Consider
\begin{lstlisting}
List.filter {α}      : (α → Bool) → List α → List α
List.filter  : ∀ {α} , (α → Bool) → List α → List α
\end{lstlisting}
    We may replace (instantiate) \lstinline{α} by any type:
\begin{lstlisting}
List.filter (α := Int)  : (Int → Bool) → List Int → List Int
@List.filter Char  : (Char → Bool) → List Char → List Char
\end{lstlisting}
  \end{block}
  \begin{block}<+->{Instantiation rule}
    \vspace{-0.5\baselineskip}
    \begin{mathpar}
      \inferrule[Inst]{e : \forall \alpha, t \\ t' : \texttt{Type}}{e : t[\alpha \mapsto t']}
    \end{mathpar}
  \end{block}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Curried functions}
\small
\begin{block}<+->{Compare these types}
\begin{lstlisting}
abbrev T1 := Int -> Int -> Int
abbrev T2 := Int × Int -> Int
\end{lstlisting}
\begin{itemize}
\item Both function types take two integers and return one
\item \texttt{T1} takes the arguments one at a time
\item \texttt{T2} takes both arguments as a pair
\end{itemize}
\end{block}
\begin{block}<+->{Lean prefers types like \texttt{T1}}
  \begin{itemize}
  \item A \textbf{curried} type, after logician \href{https://en.wikipedia.org/wiki/Haskell_Curry}{Haskell B. Curry}
  \item Predefined functions \texttt{Function.curry} and
    \texttt{Function.uncurry} map between \texttt{T1} and \texttt{T2}  (an isomorphism)
  \end{itemize}
\begin{lstlisting}
Function.curry   : (α × β → γ) → α → β → γ
Function.uncurry : (α → β → γ) → α × β → γ
\end{lstlisting}
\end{block}
\end{frame}

\begin{frame}
  \frametitle{Typing Function Applications}
  \begin{block}<+->{Typing Rule}
    \begin{mathpar}
      \inferrule[App]
      {e_1 : t_2 \to t_1 \\ e_2 : t_2}
      {(e_1\ e_2) : t_1}
    \end{mathpar}
  \end{block}
  \begin{block}<+->{Example}
    Given
    \vspace{-\baselineskip}
    \begin{eqnarray*}
      \text{map} & : & \forall\{ \alpha \beta\}, (\alpha \to \beta) \to
                        \mathtt{List}~\alpha \to List~\beta
      \\
      \text{even} & : & \mathtt{Int} \to \mathtt{Bool}
    \end{eqnarray*}
    \footnotesize
    \begin{mathpar}
      \inferrule*
      {
        \inferrule*
        {
          \inferrule*
          {\text{map}  :  \forall\{\alpha\beta\}, (\alpha \to \beta) \to \texttt{List}~\alpha \to \texttt{List}~\beta}
          { \text{map} :  (\texttt{Int} \to \texttt{Bool}) \to \texttt{List}~\texttt{Int} \to \texttt{List}~\texttt{Bool}}
          \\
          \inferrule*{}
          {\text{even} : \texttt{Int} \to \texttt{Bool}}
        }
        { \text{map}\ \text{even} : \alt<+->{\texttt{List}~\texttt{Int} \to \texttt{List}~\texttt{Bool}}{??}
        }
        \\
        \inferrule*
        {}
        {\text{[1,2,3,4,5]} : \text{List Int}}
      }
      { \text{map}\ \text{even}\ \text{[1,2,3,4,5]} : \alt<+->{\texttt{List}~\texttt{Bool}}{??}}
    \end{mathpar}
  \end{block}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Designing a higher-order function}
  \onslide<+->{}
  \begin{block}<+->{Two functions on lists}
    \begin{columns}
      \begin{column}{0.48\textwidth}
\begin{lstlisting}[basicstyle=\footnotesize]
def sum : (xs : List Nat) → Nat
  | [] => 0
  | x :: xs => x + sum xs
\end{lstlisting}
      \end{column}
      \begin{column}{0.48\textwidth}
\begin{lstlisting}
def product : (xs : List Nat) → Nat
  | [] => 1
  | x :: xs => x * sum xs
\end{lstlisting}
      \end{column}
    \end{columns}
  \end{block}
  \begin{block}<+->{The common pattern}
\begin{lstlisting}[basicstyle=\footnotesize]
def g : (xs : List α) → β
  | [] => z
  | x :: xs => s x (g xs)
\end{lstlisting}
    with parameters
    \begin{itemize}
    \item \lstinline{z : α} is a value
    \item \lstinline{s : α -> β -> β} is a combining function
    \end{itemize}
    $\Rightarrow$ Represent \texttt{g} by some \texttt{foldRight f z}
  \end{block}
\end{frame}

\begin{frame}[fragile]
  \frametitle{The foldRight function}
  \framesubtitle{Making the pattern into a higher-order function}
  \begin{block}<+->{Abstracting over value and combining function}
\begin{lstlisting}
def foldRight (s : α → β → β) (z : β) : List α → β
  | [] => z
  | x :: xs => s x (foldRight s z xs)
\end{lstlisting}
    where
    \begin{itemize}
    \item \lstinline{z : β} is a value
    \item \lstinline{s : α → β → β} is a combining function
    \end{itemize}
  \end{block}
  \begin{alertblock}<+->{What's the type of foldRight?}
    \vspace{-\baselineskip}
  \end{alertblock}
  \begin{block}<+->{}
\begin{lstlisting}
foldRight : ∀ {α β}, (α → β → β) → β → List α → β
\end{lstlisting}
  \end{block}
  % \begin{alertblock}<+->{Also known as reduce}
  %   map $+$ reduce $=$ \href{https://en.wikipedia.org/wiki/MapReduce}{MapReduce}
  % \end{alertblock}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Partial application}
\begin{lstlisting}
#check List.take
#check List.take 5

#check foldRight
#check foldRight (· + ·)
#check foldRight (· * ·) 1
\end{lstlisting}
  \begin{itemize}
  \item Partial application $=$ function application with ``too few'' arguments
  \item Result is a function
  \item Can be used like any other function
  \end{itemize}

\begin{frame}[fragile]
  \frametitle{Function composition}
  \begin{block}{Example}
    \begin{itemize}
    \item<+-> Remove spaces from string as in this
      example
\begin{lstlisting}
removeWhitespace "abc def \n ghi" == "abcdefghi"
\end{lstlisting}
    \item<+-> In module \texttt{Char}
\begin{lstlisting}
Char.isWhitespace : Char -> Bool
\end{lstlisting}
    \item<+-> yields definition
\begin{lstlisting}
def removeWhitespace (s : String) : String :=
  (s.toList.filter (not ∘ Char.isWhitespace)).toString
\end{lstlisting}
    \item<+-> Operator ``\lstinline{∘}'' is function composition defined as
      notation on top of \lstinline{Function.comp}.
\begin{lstlisting}
def compose {α β γ : Type} (f : β → γ) (g : α → β) : α → γ :=
  fun x => f (g x)
infixr :90 " ∘ " => compose
\end{lstlisting}
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Anonymous functions}
  \begin{itemize}
  \item<+-> Usual function definition
\begin{lstlisting}
def snoc (x : α) (xs : List α) : List α := xs ++ [x]
\end{lstlisting}
  \item<+->
  Alternative: define using an anonymous function aka
  \href{https://en.wikipedia.org/wiki/Anonymous_function}{lambda
    expression} 
\begin{lstlisting}
def snoc' : α → List α → List α :=
  fun x xs => xs ++ [x]
\end{lstlisting}
\item<+->
  Often for function used in one place as in
\begin{lstlisting}
def f3 : List α → List α := foldRight snoc []
\end{lstlisting}
\item<+-> Equivalently replace \texttt{snoc} by its definition
\begin{lstlisting}
def f3' : List α → List α := foldRight (fun x xs => xs ++ [x]) []
\end{lstlisting}
\item<+-> Shorten more using partial application
\begin{lstlisting}
def f3'' : List α → List α := foldRight (fun x => (· ++ [x])) []
\end{lstlisting}
\end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Eta conversion}
  \begin{enumerate}
  \item A number of definitions have the form
\begin{lstlisting}
def f (x : A) : B := g x
\end{lstlisting}
    where \texttt{x} does not occur in \texttt{g}
  \item 
    In such cases, the formal parameter \texttt{x} is redundant:
\begin{lstlisting}
def f : A -> B := g
\end{lstlisting}
    is an equivalent definition \textbf{if} \lstinline{g} is terminating.
  \item The transformation from (1) to (2) is called \textbf{eta reduction}.\footnote{Reverse transformation: \textbf{eta expansion}; both directions: \textbf{eta conversion}}
  \item The typing of an eta-reduced definition is more restricted.
\end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Wrapup}
  \begin{alertblock}{Higher-order functions}
  \begin{itemize}
  \item take functions as parameters,
  \item often have polymorphic types,
  % \item Haskell prefers curried function types.
  \item abstract common patterns (map, filter, foldr),
  \item enable powerful programming techniques (partial application, operator sections, function composition, anonymous functions, eta conversion).
  \end{itemize}
  \end{alertblock}
\end{frame}
